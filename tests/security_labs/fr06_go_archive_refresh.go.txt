// Synthetic archive-containment tests. All writes are inside t.TempDir().
// Run as an unprivileged user; no Docker API, network or host paths are used.
package archive_refresh

import (
    "archive/tar"
    "bytes"
    "os"
    "path/filepath"
    "testing"
    archive "github.com/moby/go-archive"
)

type entry struct { name, target, data string; kind byte }
func payload(t *testing.T, entries ...entry) *bytes.Reader {
    t.Helper()
    var data bytes.Buffer
    out := tar.NewWriter(&data)
    for _, item := range entries {
        header := &tar.Header{Name:item.name, Linkname:item.target, Typeflag:item.kind, Mode:0755}
        if item.kind == tar.TypeReg { header.Size=int64(len(item.data)); header.Mode=0644 }
        if err:=out.WriteHeader(header); err!=nil {t.Fatal(err)}
        if header.Size>0 {if _,err:=out.Write([]byte(item.data));err!=nil {t.Fatal(err)}}
    }
    if err:=out.Close();err!=nil {t.Fatal(err)}
    return bytes.NewReader(data.Bytes())
}
func ownedRoot(t *testing.T) (string,string) {
    t.Helper()
    if os.Geteuid()==0 {t.Fatal("unprivileged fixture runner required")}
    root:=t.TempDir();dest:=filepath.Join(root,"destination")
    if err:=os.Mkdir(dest,0755);err!=nil {t.Fatal(err)}
    return root,dest
}
func TestRegularArchivePreserved(t *testing.T) {
    _,dest:=ownedRoot(t)
    err:=archive.Untar(payload(t,entry{name:"normal.txt",data:"fixture",kind:tar.TypeReg}),dest,&archive.TarOptions{NoLchown:true})
    if err!=nil {t.Fatal(err)}
    data,err:=os.ReadFile(filepath.Join(dest,"normal.txt"));if err!=nil||string(data)!="fixture" {t.Fatalf("regular content not preserved: %v",err)}
}
func TestContainedRelativeSymlinkPreserved(t *testing.T) {
    _,dest:=ownedRoot(t)
    stream:=payload(t,entry{name:"inner",kind:tar.TypeDir},entry{name:"inner/link",target:"..",kind:tar.TypeSymlink},entry{name:"inner/link/normal.txt",data:"fixture",kind:tar.TypeReg})
    if err:=archive.Untar(stream,dest,&archive.TarOptions{NoLchown:true});err!=nil {t.Fatal(err)}
    data,err:=os.ReadFile(filepath.Join(dest,"normal.txt"));if err!=nil||string(data)!="fixture" {t.Fatalf("valid contained link rejected: %v",err)}
}
func TestChainedSymlinkCannotWriteOutsideDestination(t *testing.T) {
    root,dest:=ownedRoot(t);outside:=filepath.Join(root,"outside")
    if err:=os.Mkdir(outside,0755);err!=nil {t.Fatal(err)}
    stream:=payload(t,entry{name:"inner",kind:tar.TypeDir},entry{name:"inner/up",target:"..",kind:tar.TypeSymlink},entry{name:"inner/up/route",target:"../outside",kind:tar.TypeSymlink},entry{name:"inner/up/route/synthetic.txt",data:"fixture",kind:tar.TypeReg})
    _=archive.Untar(stream,dest,&archive.TarOptions{NoLchown:true})
    if _,err:=os.Lstat(filepath.Join(outside,"synthetic.txt"));!os.IsNotExist(err) {t.Fatalf("write escaped destination inside disposable fixture: %v",err)}
}
func TestParentTraversalCannotWriteOutsideDestination(t *testing.T) {
    root,dest:=ownedRoot(t)
    err:=archive.Untar(payload(t,entry{name:"../synthetic.txt",data:"fixture",kind:tar.TypeReg}),dest,&archive.TarOptions{NoLchown:true})
    if err==nil {t.Fatal("parent traversal was not rejected")}
    if _,err:=os.Lstat(filepath.Join(root,"synthetic.txt"));!os.IsNotExist(err) {t.Fatal("parent traversal changed fixture sibling")}
}
